# Python voice agent — install with:  pip install -r requirements.txt
websockets        # agent.py — WebSocket server the browser app connects to
requests          # intent.py — talks to the local Ollama LLM API

# Only needed for the `voice <file.ogg>` command in agent.py:
faster-whisper    # local speech-to-text (first run downloads the ~1.5 GB model)

# Microphone recording (the `mic` command in agent.py):
sounddevice       # records from the default microphone via PortAudio
scipy             # writes the recorded audio as a WAV file

# Text-to-speech (auto-plays a short Urdu confirmation after each highlight):
edge-tts          # Microsoft neural TTS with good Urdu voice support
